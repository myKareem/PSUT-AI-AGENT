# ============================================================
#  Arabic STT Service — pinned dependencies
#  Tested on: Windows 11, Python 3.11, CUDA 12.4
# ============================================================

# ── Web framework ────────────────────────────────────────────
fastapi==0.115.6
uvicorn[standard]==0.32.1     # uvloop not available on Windows;
                               # [standard] installs websockets + httptools
python-multipart==0.0.20       # required for multipart/form-data uploads

# ── Audio I/O ────────────────────────────────────────────────
# torchaudio must match the torch version EXACTLY.
# Both are installed together via the torch index URL below.
torchaudio==2.5.1
soundfile==0.12.1              # libsndfile binding — fast WAV encode/decode
                               # (used only for direct WAV inspection; the
                               #  main pipeline bypasses WAV entirely)

# ── Deep learning runtime ────────────────────────────────────
#
# CPU-only (default — works on any Windows 11 machine):
torch==2.5.1
#
# ── GPU / CUDA users — replace the torch line above with: ────
#
#   CUDA 12.4 (RTX 30xx / 40xx — most common on Windows 11):
#     pip install torch==2.5.1+cu124 torchaudio==2.5.1+cu124 \
#         --index-url https://download.pytorch.org/whl/cu124
#
#   CUDA 11.8 (RTX 20xx / older):
#     pip install torch==2.5.1+cu118 torchaudio==2.5.1+cu118 \
#         --index-url https://download.pytorch.org/whl/cu118
#
#   After installing the CUDA torch build, also install the
#   matching cuDNN and CUDA toolkit from NVIDIA's site, OR use
#   the bundled wheels:
#     pip install nvidia-cudnn-cu12==9.1.0.70
#
# ── faster-whisper (CTranslate2 backend) ─────────────────────
faster-whisper==1.1.0
# CTranslate2 is a transitive dependency of faster-whisper.
# On Windows it ships as a pre-built wheel with CPU support.
# For CUDA support the CUDA wheel is pulled in automatically
# when torch with CUDA is already installed in the environment.

# ── Utilities ────────────────────────────────────────────────
numpy==2.1.3                   # float32 array bridge between torch and whisper
